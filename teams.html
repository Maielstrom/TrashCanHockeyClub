   <script>

       // Placeholder JavaScript for dropdowns. This will handle showing/hiding the menus on click/hover.

       const teamButton = document.getElementById('teams-menu-button');

       const teamsMenu = document.getElementById('teams-menu');

       const merchButton = document.getElementById('merch-menu-button');

       const merchMenu = document.getElementById('merch-menu');


       function toggleDropdown(menu) {

           menu.classList.toggle('hidden');

       }


       teamButton.addEventListener('click', (e) => {

           e.preventDefault();

           toggleDropdown(teamsMenu);

       });


       merchButton.addEventListener('click', (e) => {

           e.preventDefault();

           toggleDropdown(merchMenu);

       });


       // Hide menus if user clicks outside of them

       window.addEventListener('click', (e) => {

           if (!teamButton.contains(e.target) && !teamsMenu.contains(e.target)) {

               teamsMenu.classList.add('hidden');

           }

           if (!merchButton.contains(e.target) && !merchMenu.contains(e.target)) {

               merchMenu.classList.add('hidden');

           }

       });
