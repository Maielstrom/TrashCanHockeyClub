   <!-- Simplified JavaScript with Firebase -->

   <script type="module">

       import { initializeApp } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-app.js";

       import {

           getFirestore,

           doc,

           setDoc,

           collection,

           onSnapshot,

           getDocs,

           query

       } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-firestore.js";

       import { getAuth, signInAnonymously, signInWithCustomToken } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-auth.js";

      

       // This function will create and return a table row HTML string

       function createPlayerRow(player) {

           return `

               <tr class="hover:bg-gray-700 transition-colors">

                   <td class="px-4 py-2">${player.number}</td>

                   <td class="px-4 py-2">${player.name}</td>

                   <td class="px-4 py-2">${player.position}</td>

               </tr>

           `;

       }


       window.onload = async function() {

           try {

               // IMPORTANT: When you launch your own site, you will need to replace the following

               // two variables with your own Firebase project credentials.

               // You can get these from your Firebase Console.

               const firebaseConfig = JSON.parse(typeof __firebase_config !== 'undefined' ? __firebase_config : '{}');

               const authCustomToken = typeof __initial_auth_token !== 'undefined' ? __initial_auth_token : null;

               const appId = typeof __app_id !== 'undefined' ? __app_id : 'default-app-id';


               const app = initializeApp(firebaseConfig);

               const db = getFirestore(app);

               const auth = getAuth(app);

              

               // Sign in the user. If you are using your own firebase project and

               // a different authentication method, you will need to update this line.

               if (authCustomToken) {

                   await signInWithCustomToken(auth, authCustomToken);

               } else {

                   await signInAnonymously(auth);

               }

              

               // Use the application ID to create a unique path for the database.

               const teamPlayersCollectionPath = `/artifacts/${appId}/public/data/brewster_dumpster_fire_roster`;

              

               const playersRef = collection(db, teamPlayersCollectionPath);


               // Check if the collection is empty. If so, populate it with some sample data.

               const snapshot = await getDocs(playersRef);

               if (snapshot.empty) {

                   console.log("No data found. Populating with sample data.");

                  

                   const samplePlayers = [

                       { name: "Bill Jones", number: 10, position: "Forward" },

                       { name: "Mike Davis", number: 44, position: "Defense" },

                       { name: "Sam Williams", number: 35, position: "Goalie" }

                   ];


                   for (const player of samplePlayers) {

                       await setDoc(doc(playersRef), player);

                   }

               } else {

                   console.log("Collection already populated. Skipping sample data creation.");

               }


               // Set up the real-time listener to read the data

               onSnapshot(query(playersRef), (querySnapshot) => {

                   const rosterTableBody = document.getElementById('roster-table-body');

                   const loadingText = document.getElementById('roster-loading-text');


                   rosterTableBody.innerHTML = ''; // Clear previous roster

                   loadingText.style.display = 'none'; // Hide loading text


                   querySnapshot.forEach((doc) => {

                       const playerData = doc.data();

                       rosterTableBody.innerHTML += createPlayerRow(playerData);

                   });


                   console.log("Roster data successfully loaded from Firestore.");

               });


               // Slideshow logic

               let slideIndex = 1;

               showSlides(slideIndex);


               function plusSlides(n) {

                   showSlides(slideIndex += n);

               }


               function showSlides(n) {

                   let i;

                   let slides = document.getElementsByClassName("mySlides");

                   if (n > slides.length) {slideIndex = 1}   

                   if (n < 1) {slideIndex = slides.length}

                   for (i = 0; i < slides.length; i++) {

                       slides[i].style.display = "none"; 

                   }

                   slides[slideIndex-1].style.display = "block"; 

               }


               window.plusSlides = plusSlides;


           } catch (error) {

               console.error("Failed to connect to Firebase or read documents:", error);

           }

       };

   </script>
